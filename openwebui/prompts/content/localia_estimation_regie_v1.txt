Schema JSON de sortie:
{

  "temps_par_tache": [
    {
      "tache": "string",
      "quantite": "number|null",
      "unite": "m2|ml|h|forfait|null",

      "justification": "string"
    }
  ],
  "hypotheses": ["string"],
  "questions_bloquantes": ["string"],

}
